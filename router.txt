A. Konfigurasi Router (Gateway & DHCP Server)

enable
configure terminal

! 1. Memberikan IP pada port yang mengarah ke Switch
interface fastEthernet 0/0
 ip address 192.224.10.1 255.255.255.192
 no shutdown
 exit

! 2. Membuat DHCP Server
ip dhcp excluded-address 192.224.10.1
ip dhcp pool LOKAL_LAN
 network 192.224.10.0 255.255.255.192
 default-router 192.224.10.1
 exit

! Simpan konfigurasi
write memory

B. Konfigurasi Core Switch

enable
configure terminal
hostname Core_Switch

! 1. Mengubah port 22, 23, 24 menjadi jalur Trunk (antar perangkat)
interface range fastEthernet 0/22 - 24
 switchport mode trunk
 exit

! 2. Mengaktifkan PC agar cepat mendapat IP (Portfast) di port 1-20
interface range fastEthernet 0/1 - 20
 switchport mode access
 spanning-tree portfast
 exit

! Simpan konfigurasi
write memory

C. Konfigurasi Extension Switch 1 & 2

enable
configure terminal
hostname Ext_Switch

! 1. Mengubah port 24 menjadi jalur Trunk (mengarah ke Core Switch)
interface fastEthernet 0/24
 switchport mode trunk
 exit

! 2. Mengaktifkan PC agar cepat mendapat IP (Portfast) di port 1-20
interface range fastEthernet 0/1 - 20
 switchport mode access
 spanning-tree portfast
 exit

! Simpan konfigurasi
write memory
